  d.setMinutes(Math.ceil(d.getMinutes() / 15) * 15, 0, 0)
  return { time: d.toTimeString().slice(0, 5), demain: d.getDate() !== now.getDate() }
}

                    <button onClick={() => setAgeOn(v => !v)} style={{ display: 'flex', alignItems: 'center', gap: 8, background: 'transparent', border: 'none', padding: 0, cursor: 'pointer', fontFamily: 'inherit', marginBottom: ageOn ? 8 : 0 }}>
                      <span style={labelS as any}>🔞 {en ? 'AGE LIMIT' : 'LIMITE D’ÂGE'} <span style={{ fontWeight: 600 }}>· {en ? 'optional' : 'optionnel'}</span></span>
                      <span style={{ width: 34, height: 19, borderRadius: 10, background: ageOn ? P.green : P.border, position: 'relative', transition: '.15s', flexShrink: 0, display: 'inline-block' }}>
                        <span style={{ position: 'absolute', top: 2, left: ageOn ? 17 : 2, width: 15, height: 15, borderRadius: '50%', background: '#fff', transition: '.15s' }} />
                      </span>
                    </button>
                    {/* 🎡 JogWheels (David 14.07 : « les mêmes molettes que quand on choisit l'âge » — c'est
                        littéralement le même composant que la feuille d'âge). Cross-ajustement min/max. */}
                    {ageOn && (
                      <div style={{ display: 'flex', alignItems: 'center', gap: 12 }}>
                        <div style={{ flex: 1 }}>
                          <div style={{ fontSize: 10.5, fontWeight: 800, color: P.ink, textAlign: 'center', marginBottom: 4 }}>{en ? 'from' : 'de'}</div>
                          <JogWheel slots={AGE_SLOTS} value={String(ageMin)} onChange={v => { const n = parseInt(v); setAgeMin(n); if (n > ageMax) setAgeMax(n) }} accent />
                        </div>
                        <div style={{ fontSize: 16, color: P.ink, fontWeight: 800 }}>→</div>
                        <div style={{ flex: 1 }}>
                          <div style={{ fontSize: 10.5, fontWeight: 800, color: P.ink, textAlign: 'center', marginBottom: 4 }}>{en ? 'to' : 'à'}</div>
                          <JogWheel slots={AGE_SLOTS} value={String(ageMax)} onChange={v => { const n = parseInt(v); setAgeMax(n); if (n < ageMin) setAgeMin(n) }} />
                        </div>
                      </div>
                    )}
                  </div>
                  {/* Niveau */}
                  <div style={{ marginBottom: 14 }}>
                    <div style={labelS}>{en ? 'LEVEL' : 'NIVEAU'} <span style={{ fontWeight: 600 }}>· {en ? 'optional' : 'optionnel'}</span></div>
                    <div style={{ display: 'flex', gap: 6 }}>
                      {([[null, en ? 'Everyone' : 'Tous'], ['debutant', en ? 'Beginner' : 'Débutant'], ['moyen', en ? 'Casual' : 'Moyen'], ['avance', en ? 'Advanced' : 'Avancé']] as const).map(([k, l]) => (
                        <button key={String(k)} onClick={() => setLevel(k)} style={{ ...chipS(level === k), padding: '8px 0', fontSize: 11.5 }}>{l}</button>
                      ))}
                    </div>
                  </div>
                  {/* Prix */}
                  <div style={{ marginBottom: 6 }}>
                    <div style={labelS}>💰 {en ? 'PRICE' : 'PRIX'}</div>
                    <div style={{ display: 'flex', gap: 6, marginBottom: 7 }}>
                      {[en ? 'Free' : 'Gratuit', '5 CHF', '10 CHF', '20 CHF'].map(pz => (
                        <button key={pz} onClick={() => setPrice(pz)} style={{ ...chipS(price === pz), padding: '8px 0', fontSize: 11.5 }}>{pz}</button>
                      ))}
                    </div>
                    <input value={price} onChange={e => setPrice(e.target.value.slice(0, 40))} placeholder={en ? 'Or another price… (default: Free)' : 'Ou un autre prix… (par défaut : Gratuit)'} style={{ ...inputS(false), fontSize: 13, padding: '10px 13px' }} />
                  </div>
                </div>
              )}
            </div>
          )})()}

          {/* 🧱 NAVIGATION IN-FLOW (doublon volontaire du footer — bug David : footer invisible sur son iPhone).
              En fin de contenu défilable → TOUJOURS atteignable en scrollant, quel que soit le viewport. */}
          <div style={{ display: 'flex', gap: 8, marginTop: 18, paddingBottom: 6 }}>
            {step > 1 && (
              <button onClick={goPrev} style={{ flex: '0 0 auto', padding: '13px 16px', borderRadius: 14, border: `1.5px solid ${P.border}`, background: 'transparent', color: P.plum, fontSize: 14, fontWeight: 800, cursor: 'pointer', fontFamily: 'inherit' }}>
                ← {en ? 'Back' : 'Retour'}
              </button>
            )}
            {step < 2 ? (
              <button onClick={goNext} style={{ flex: 1, padding: '13px', borderRadius: 14, border: 'none', background: stepDone[step] ? P.pink : `${P.plum}26`, color: stepDone[step] ? '#fff' : P.ink, fontSize: 15, fontWeight: 900, cursor: 'pointer', fontFamily: 'inherit' }}>
                {en ? 'Continue' : 'Continuer'} →
              </button>
            ) : (
              <button onClick={publish} disabled={creating} style={{ flex: 1, padding: '13px', borderRadius: 14, border: 'none', background: P.pink, color: '#fff', fontSize: 15, fontWeight: 900, cursor: 'pointer', fontFamily: 'inherit' }}>
                {creating ? (en ? 'Creating…' : 'Création…') : (en ? '✦ Publish event' : '✦ Publier l\'événement')}
              </button>
            )}
          </div>
        </div>

        {/* Pied fixe : navigation des gestes */}
        <div style={{ flexShrink: 0, padding: '12px 20px calc(var(--sab, 0px) + 16px)', borderTop: `1px solid ${P.border}`, background: P.white, display: 'flex', gap: 8 }}>
          {step > 1 && (
            <button onClick={goPrev} style={{ flex: '0 0 auto', padding: '14px 18px', borderRadius: 14, border: `1.5px solid ${P.border}`, background: 'transparent', color: P.plum, fontSize: 14, fontWeight: 800, cursor: 'pointer', fontFamily: 'inherit' }}>
              ← {en ? 'Back' : 'Retour'}
            </button>
          )}
          {step < 2 ? (
            <button onClick={goNext}
              style={{ flex: 1, padding: '14px', borderRadius: 14, border: 'none', background: stepDone[step] ? P.pink : `${P.plum}26`, color: stepDone[step] ? '#fff' : P.ink, fontSize: 15, fontWeight: 900, cursor: 'pointer', fontFamily: 'inherit', transition: '.2s' }}>
              {en ? 'Continue' : 'Continuer'} →
            </button>
          ) : (
            <button onClick={publish} disabled={creating}
              style={{ flex: 1, padding: '14px', borderRadius: 14, border: 'none', background: P.pink, color: '#fff', fontSize: 15, fontWeight: 900, cursor: 'pointer', fontFamily: 'inherit' }}>
              {creating ? (en ? 'Creating…' : 'Création…') : (en ? '✦ Publish event' : '✦ Publier l\'événement')}
            </button>
          )}
        </div>

        {/* Confirmation avant de quitter si commencé (même garde-fou que V1) */}
        {confirmClose && (
          <div style={{ position: 'absolute', inset: 0, background: 'rgba(74,42,61,.5)', backdropFilter: 'blur(4px)', display: 'flex', alignItems: 'center', justifyContent: 'center', padding: 24 }}>
            <div style={{ background: P.white, borderRadius: 18, padding: '20px 18px', maxWidth: 300, textAlign: 'center', boxShadow: '0 20px 50px rgba(83,41,67,.4)' }}>
              <div style={{ fontSize: 15, fontWeight: 900, color: P.plum, marginBottom: 6 }}>{en ? 'Leave without publishing?' : 'Quitter sans publier ?'}</div>
              <div style={{ fontSize: 12, color: P.ink, lineHeight: 1.5, marginBottom: 16 }}>{en ? 'You started filling in the event. Everything will be lost.' : 'Tu as commencé à remplir l\'événement. Tout sera perdu.'}</div>
              <div style={{ display: 'flex', gap: 8 }}>
                <button onClick={() => setConfirmClose(false)} style={{ flex: 1, padding: '11px', borderRadius: 11, border: `1px solid ${P.border}`, background: 'transparent', color: P.plum, fontSize: 13, fontWeight: 800, cursor: 'pointer', fontFamily: 'inherit' }}>{en ? 'Continue' : 'Continuer'}</button>
                <button onClick={onClose} style={{ flex: 1, padding: '11px', borderRadius: 11, border: 'none', background: P.plum, color: '#fff', fontSize: 13, fontWeight: 800, cursor: 'pointer', fontFamily: 'inherit' }}>{en ? 'Leave' : 'Quitter'}</button>
              </div>
            </div>
          </div>
        )}
      </div>
    </div>
  )
}
